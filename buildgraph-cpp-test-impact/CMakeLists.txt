cmake_minimum_required(VERSION 3.16)
project(buildgraph LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(buildgraph src/main.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(buildgraph PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()
enable_testing()
add_test(NAME stats COMMAND buildgraph ${CMAKE_CURRENT_SOURCE_DIR}/data/modules.csv ${CMAKE_CURRENT_SOURCE_DIR}/data/tests.csv stats)
add_test(NAME impact COMMAND buildgraph ${CMAKE_CURRENT_SOURCE_DIR}/data/modules.csv ${CMAKE_CURRENT_SOURCE_DIR}/data/tests.csv impact module-7)
add_test(NAME cycles COMMAND buildgraph ${CMAKE_CURRENT_SOURCE_DIR}/data/modules.csv ${CMAKE_CURRENT_SOURCE_DIR}/data/tests.csv cycles)
